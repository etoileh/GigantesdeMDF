342 if (v > 255) v = 255;

447void configurarBotoes() {
448 uint8_t mask =
449 (1 << BTN_FRENTE) |
450 (1 << BTN_TRAS) |
451 (1 << BTN_ESQUERDA) |
452 (1 << BTN_DIREITA);
453
454 DDRC &= ~mask;
455 PORTC |= mask;
456
457 DDRB &= ~(1 << BTN_RESET);
458 PORTB |= (1 << BTN_RESET);
459
460 uint8_t leituraC = PINC;
461 uint8_t leituraB = PINB;
462
463 for (uint8_t i = 0; i < 5; i++) {
464 uint8_t valor;
465
466 if (i == 4) {
467 valor = (leituraB & mascaras[i]) ? 1 : 0;
468 } else {
469 valor = (leituraC & mascaras[i]) ? 1 : 0;
470 }
471
472 brutoAnterior[i] = valor;
473 estavel[i] = valor;
474 desdeMudanca[i] = 0;
475 }
476}

489uint8_t atualizarBotao(
490 uint8_t indice,
491 uint8_t leitura,
492 uint32_t agora
493) {
494 uint16_t debounce =
495 (indice == 4) ?
496 DEBOUNCE_RESET_MS :
497 DEBOUNCE_MS;
498
499 if (leitura != brutoAnterior[indice]) {
500 brutoAnterior[indice] = leitura;
501 desdeMudanca[indice] = agora;
502 }
503
504 if (leitura != estavel[indice] &&
505 (uint32_t)(agora - desdeMudanca[indice]) >= debounce) {
506
507 estavel[indice] = leitura;
508 return 1;
509 }
510
511 return 0;
512}

520uint8_t atualizarBotoes(uint32_t agora) {
521 uint8_t leituraC = PINC;
522 uint8_t leituraB = PINB;
523
524 for (uint8_t i = 0; i < 4; i++) {
525 uint8_t valor =
526 (leituraC & mascaras[i]) ? 1 : 0;
527
528 atualizarBotao(i, valor, agora);
529 }
530
531 uint8_t reset =
532 (leituraB & mascaras[4]) ? 1 : 0;
533
534 return atualizarBotao(4, reset, agora);

593uint8_t verificarLDR(uint32_t agora) {
594 if ((uint32_t)(agora - ultimaAmostraLDR) <
595 LDR_INTERVALO_MS) {
596 return 0;
597 }
598
599 ultimaAmostraLDR = agora;
600 valorLDR = lerADC();
601
602 if (valorLDR >= LDR_LIMIAR_DISPARO) {
603 if (contadorDano < LDR_CONFIRMACOES) {
604 contadorDano++;
605 }
606
607 if (contadorDano >= LDR_CONFIRMACOES) {
608 contadorDano = 0;
609 return 1;
610 }
611 } else {
612 contadorDano = 0;
613 }
614
615 return 0;
616}

638
639 PORTD = (PORTD & ~mask) | saida;

830void loop() {
831 static uint32_t ultimoSerial = 0;
832
833 uint32_t agora = agoraMs();
834
835 uint8_t mudouReset = atualizarBotoes(agora);
836
837 switch (estado) {
838
839 case ATIVO: {
840 controleManual();
841
842 if (verificarLDR(agora)) {
843 receberDano(agora);
844 break;
845 }
846
847 if ((uint32_t)(agora - ultimoSerial) >=
848 SERIAL_INTERVALO_MS) {
849
850 ultimoSerial = agora;
851
852 uartTexto("LDR = ");
853 uartNumero(valorLDR);
854
855 uartTexto(" | Confirmacoes = ");
856 uartNumero(contadorDano);
857
858 uartTexto(" | Vidas = ");
859 uartNumero(vidas);
860
861 uartTexto("\r\n");
862 }
863
864 break;
865 }
866
867 case ATINGIDO: {
868 uint32_t dt = agora - inicioDano;
869
870 if (dt < TEMPO_GIRO_MS) {
871 motores(VELOCIDADE, -VELOCIDADE);
872 } else {
873 pararMotores();
874 }
875
876 // Sem leituras ADC durante a protecao
877 if (dt >= TEMPO_PROTECAO_MS) {
878 pararMotores();
879
880 if (vidas == 0) {
881 estado = ELIMINADO;
882 uartLinha(
883 "ELIMINADO - PRESSIONE RESET"
884 );
885 } else {
886 contadorDano = 0;
887 ultimaAmostraLDR = agora;
888
889 estado = ATIVO;
890 uartLinha("PROTECAO TERMINOU");
891 }
892 }
893
894 break;
895 }
896
897 case ELIMINADO: {
898 pararMotores();
899
900 if (mudouReset && estavel[4] == 0) {
901 reiniciar(agora);
902 }
903
904 break;
905 }
906 }
